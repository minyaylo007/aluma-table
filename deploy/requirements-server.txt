# ЧИМ ЦЕЙ ФАЙЛ ВІДРІЗНЯЄТЬСЯ ВІД api/requirements.txt
# ═══════════════════════════════════════════════════════════════════════════
#
# api/requirements.txt — це список для ЛЯМБДИ, і там рівно один рядок:
#
#     requests>=2.31
#
# boto3 у ньому нема НЕ через недогляд: у середовищі AWS Lambda boto3 уже
# лежить у рантаймі, ставити його в пакет означало б тягнути ~50 МБ дарма.
#
# На своєму сервері рантайму AWS нема, і boto3 треба ставити явно — НАВІТЬ
# ЯКЩО МИ ПРАЦЮЄМО В РЕЖИМІ SQLITE І ДО DYNAMODB НЕ ЗВЕРТАЄМОСЬ ЖОДНОГО РАЗУ.
# Причина в тому, що api/handler.py:33 імпортує його БЕЗУМОВНО:
#
#     import boto3
#     from boto3.dynamodb.conditions import Key
#
# Цей import стоїть на 33-му рядку, тобто ВИЩЕ за читання FX_STORAGE
# (handler.py:63). Отже вибір сховища відбувається вже після імпорту, і
# «ми ж на sqlite» від ImportError не рятує: без boto3 модуль не імпортується
# зовсім, gunicorn падає на старті з ModuleNotFoundError і юніт іде в
# перезапуск по колу. Симптом на боці користувача — 502 від nginx на всьому
# /api/fx/ і на /admin, при живому сайті зі статики.
#
# Другий бік того самого: boto3 тут потрібен ще й для scripts/migrate_fx.py
# --phase dump, який читає (тільки читає) бойовий DynamoDB.
#
# ═══════════════════════════════════════════════════════════════════════════
# ЧОМУ ВЕРСІЇ ПРИБИТІ ЧЕРЕЗ `==`, А НЕ `>=`
# ═══════════════════════════════════════════════════════════════════════════
#
# Це рівно ті версії, на яких прогнані 116 тестів (`Ran 116 tests ... OK`)
# у /tmp/aluma-venv на Python 3.12.3. `>=` означає «яка сьогодні вийшла» —
# тобто прод поїде на комбінації, якої ніхто ніколи не запускав. Оновлення
# версії тут — окрема свідома правка з повторним прогоном тестів.
#
# Установка на сервері (ТІЛЬКИ прод-частина):
#     /opt/ihor/aluma-table/.venv/bin/pip install -r deploy/requirements-server.txt
#
# Транзитивні залежності (botocore, urllib3, s3transfer, certifi, …) навмисно
# не перелічені: їх тягне pip за прибитими прямими. Якщо колись знадобиться
# побайтово відтворити оточення — знімати `pip freeze` з робочого venv, а не
# дописувати їх сюди руками.

# WSGI-сервер перед api/wsgi.py:application.
# Запускається з deploy/systemd/aluma-api.service (-b 127.0.0.1:8005).
#
# Юніт запускає gunicorn з --no-control-socket (чому — див. коментар у
# юніті: керуючий сокет один на користувача, а сайт живе під тим самим ihor).
# Цей прапорець зʼявився в gunicorn 25.1.0
# (https://github.com/benoitc/gunicorn/blob/master/docs/content/news.md,
# розділ 25.1.0, PR #3505). На старішій версії gunicorn не знає прапорця,
# падає на старті з «unrecognized arguments», і юніт іде в перезапуск по колу.
# Пин нижче лежить усередині дозволеного діапазону >=25.1,<27.
# ОНОВЛЮЄШ gunicorn ВИЩЕ 26.x — спершу перевір, що прапорець так і зветься:
#     .venv/bin/gunicorn --help | grep -- --no-control-socket
gunicorn==26.2.0

# Див. великий блок вище: потрібен НАВІТЬ у режимі sqlite, бо handler.py:33
# імпортує його безумовно. Прибрати не можна, поки цей import безумовний.
boto3==1.34.46

# Сповіщення власнику в Telegram — notify() у handler.py.
# handler.py:36-39 ловить ImportError і тоді просто мовчить, тобто без requests
# сервіс підніметься, але заявки перестануть приходити в телеграм НЕПОМІТНО.
# Тому це прод-залежність, а не «за бажанням».
requests==2.31.0
